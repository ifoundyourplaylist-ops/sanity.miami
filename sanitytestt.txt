-- Draggable watermark, but keep size constant

    local dragging, dragStart, startPos
    Watermark.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = Watermark.Position
        end
    end)
    Watermark.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            Watermark.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
            Watermark.Size = UDim2.fromOffset(200, 20) -- enforce fixed size
        end
    end)
end

local function createToggle(parent, text)
    local container = Instance.new("Frame")
    container.BackgroundTransparency = 1
    container.Size = UDim2.new(1, -20, 0, 24)
    container.LayoutOrder = 2
    container.Parent = parent

    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Position = UDim2.fromOffset(0, 0)
    label.Size = UDim2.new(1, -40, 1, 0)
    label.Font = Enum.Font.Code
    label.Text = text
    label.TextColor3 = Color3.fromRGB(210, 210, 210)
    label.TextSize = 14
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = container

    local toggle = Instance.new("TextButton")
    toggle.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    toggle.BorderSizePixel = 0
    toggle.Size = UDim2.fromOffset(32, 18)
    toggle.Position = UDim2.new(1, -32, 0.5, 0)
    toggle.AnchorPoint = Vector2.new(1, 0.5)
    toggle.Font = Enum.Font.Code
    toggle.Text = ""
    toggle.Parent = container

    local tCorner = Instance.new("UICorner", toggle)
    tCorner.CornerRadius = UDim.new(1, 0)

    local knob = Instance.new("Frame")
    knob.BackgroundColor3 = Color3.fromRGB(200, 200, 200)
    knob.BorderSizePixel = 0
    knob.Size = UDim2.fromOffset(14, 14)
    knob.Position = UDim2.fromOffset(2, 2)
    knob.Parent = toggle

    local kCorner = Instance.new("UICorner", knob)
    kCorner.CornerRadius = UDim.new(1, 0)

    local state = false
    local function update()
        if state then
            toggle.BackgroundColor3 = Color3.fromRGB(220, 220, 220)
            knob.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
            knob.Position = UDim2.fromOffset(32 - 14 - 2, 2)
        else
            toggle.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
            knob.BackgroundColor3 = Color3.fromRGB(200, 200, 200)
            knob.Position = UDim2.fromOffset(2, 2)
        end
    end

    toggle.MouseButton1Click:Connect(function()
        state = not state
        update()
    end)

    update()
    return {
        Set = function(v)
            state = v
            update()
        end,
        Get = function()
            return state
        end,
    }
end

local function createSlider(parent, text, default)
    local container = Instance.new("Frame")
    container.BackgroundTransparency = 1
    container.Size = UDim2.new(1, -20, 0, 32)
    container.LayoutOrder = 2
    container.Parent = parent

    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Position = UDim2.fromOffset(0, 0)
    label.Size = UDim2.new(1, 0, 0, 16)
    label.Font = Enum.Font.Code
    label.Text = text
    label.TextColor3 = Color3.fromRGB(210, 210, 210)
    label.TextSize = 14
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = container

    local bar = Instance.new("Frame")
    bar.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
    bar.BorderSizePixel = 0
    bar.Position = UDim2.fromOffset(0, 18)
    bar.Size = UDim2.new(1, -40, 0, 6)
    bar.Parent = container

    local barCorner = Instance.new("UICorner", bar)
    barCorner.CornerRadius = UDim.new(1, 0)

    local fill = Instance.new("Frame")
    fill.BackgroundColor3 = Color3.fromRGB(220, 220, 220)
    fill.BorderSizePixel = 0
    fill.Size = UDim2.new((default or 0), 0, 1, 0)
    fill.Parent = bar

    local fillCorner = Instance.new("UICorner", fill)
    fillCorner.CornerRadius = UDim.new(1, 0)

    local valueLabel = Instance.new("TextLabel")
    valueLabel.BackgroundTransparency = 1
    valueLabel.Position = UDim2.new(1, -40, 0, 14)
    valueLabel.Size = UDim2.fromOffset(40, 16)
    valueLabel.Font = Enum.Font.Code
    valueLabel.Text = tostring(math.floor((default or 0) * 100)) .. "%"
    valueLabel.TextColor3 = Color3.fromRGB(200, 200, 200)
    valueLabel.TextSize = 12
    valueLabel.TextXAlignment = Enum.TextXAlignment.Right
    valueLabel.Parent = container

    local value = default or 0

    local function setFromX(x)
        local rel = math.clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        value = rel
        fill.Size = UDim2.new(rel, 0, 1, 0)
        valueLabel.Text = tostring(math.floor(rel * 100)) .. "%"
    end

    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            setFromX(input.Position.X)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
            if bar.AbsolutePosition and bar.AbsoluteSize then
                setFromX(input.Position.X)
            end
        end
    end)

    return {
        Get = function()
            return value
        end,
        Set = function(v)
            value = math.clamp(v, 0, 1)
            fill.Size = UDim2.new(value, 0, 1, 0)
            valueLabel.Text = tostring(math.floor(value * 100)) .. "%"
        end,
    }
end

local CONFIG_DIR = "sanity_configs"
local Configs = {}
local CurrentAutoload = "sanity_default"

local function safeWrite(path, data)
    local ok, err = pcall(function()
        if writefile then
            writefile(path, data)
        end
    end)
    return ok, err
end

local function safeRead(path)
    if not isfile or not isfile(path) then return nil end
    local ok, result = pcall(function()
        return readfile(path)
    end)
    if ok then
        return result
    end
    return nil
end

local function listConfigFiles()
    local names = {}
    if listfiles then
        local ok, files = pcall(function()
            return listfiles(CONFIG_DIR)
        end)
        if ok and files then
            for _, f in ipairs(files) do
                local name = f:match("([^/\\]+)%.json$")
                if name then
                    table.insert(names, name)
                end
            end
        end
    end
    return names
end

local function refreshConfigList()
    local names = listConfigFiles()
    table.sort(names)
    if #names == 0 then
        ListDropdown.Text = "---"
    else
        ListDropdown.Text = names[1]
    end
end

local function getCurrentConfigTable()
    return {
        ShowWatermark = ToggleWatermark.Get(),
        CustomCursor = ToggleCursor.Get(),
        NotificationSide = DropdownNotifSide.Text,
        DPIScale = SliderDPI.Get(),
        HitboxEnabled = nil, -- filled later
        HitboxOthersOnly = nil,
    }
end

local function applyConfigTable(tbl)
    if tbl.ShowWatermark ~= nil then
        ToggleWatermark.Set(tbl.ShowWatermark)
    end
    if tbl.CustomCursor ~= nil then
        ToggleCursor.Set(tbl.CustomCursor)
    end
    if tbl.NotificationSide then
        DropdownNotifSide.Text = tbl.NotificationSide
    end
    if tbl.DPIScale ~= nil then
        SliderDPI.Set(tbl.DPIScale)
        UIScale.Scale = 0.75 + tbl.DPIScale * 0.75
    end
end

local function saveConfig(name)
    if name == "" then name = "config_" .. tostring(os.time()) end
    local path = CONFIG_DIR .. "/" .. name .. ".json"
    local data = getCurrentConfigTable()
    local encoded = HttpService:JSONEncode(data)
    safeWrite(path, encoded)
    refreshConfigList()
    ListDropdown.Text = name
end

local function loadConfig(name)
    local path = CONFIG_DIR .. "/" .. name .. ".json"
    local raw = safeRead(path)
    if not raw then return end
    local ok, decoded = pcall(function()
        return HttpService:JSONDecode(raw)
    end)
    if ok and decoded then
        applyConfigTable(decoded)
    end
end

local HitboxBoxes = {}

local function clearHitboxes()
    for plr, boxes in pairs(HitboxBoxes) do
        for _, box in ipairs(boxes) do
            box:Destroy()
        end
    end
    HitboxBoxes = {}
end

local function createHitboxesForCharacter(char)
    local boxes = {}
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            local sb = Instance.new("SelectionBox")
            sb.Adornee = part
            sb.LineThickness = 0.03
            sb.Color3 = Color3.fromRGB(0, 255, 0)
            sb.SurfaceColor3 = Color3.fromRGB(0, 255, 0)
            sb.SurfaceTransparency = 0.8
            sb.Parent = ROOT
            table.insert(boxes, sb)
        end
    end
    return boxes
end

local function updateHitboxes()
    clearHitboxes()
    if not HitboxToggle.Get() then return end

    for _, plr in ipairs(Players:GetPlayers()) do
        if HitboxOthersOnlyToggle.Get() and plr == LOCAL_PLAYER then
            -- skip local
        else
            local char = plr.Character
            if char then
                HitboxBoxes[plr] = createHitboxesForCharacter(char)
            end
        end
    end
end

local function tweenTransparency(frame, target, duration)
    for _, obj in ipairs(frame:GetDescendants()) do
        if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
            TweenService:Create(
                obj,
                TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                {TextTransparency = target}
            ):Play()
        elseif obj:IsA("Frame") or obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
            TweenService:Create(
                obj,
                TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                {BackgroundTransparency = target}
            ):Play()
        end
    end
    TweenService:Create(
        frame,
        TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {BackgroundTransparency = target}
    ):Play()
end
