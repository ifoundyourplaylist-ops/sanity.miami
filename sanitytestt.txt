local function ensureFolder(path)
    if not isfolder(path) then
        makefolder(path)
    end
end

local function saveConfig(name)
    ensureFolder(CONFIG_FOLDER)
    name = name or CurrentConfigName
    local path = CONFIG_FOLDER .. "/" .. name .. ".json" [cite: 5]
    writefile(path, HttpService:JSONEncode(CurrentConfig))
end

local function loadConfig(name)
    ensureFolder(CONFIG_FOLDER)
    name = name or CurrentConfigName
    local path = CONFIG_FOLDER .. "/" .. name .. ".json"
    if isfile(path) then
        local ok, data = pcall(function()
            return HttpService:JSONDecode(readfile(path))
        end)
        if ok and type(data) == "table" then
            CurrentConfig = data
            CurrentConfigName = name [cite: 6]
            return true
        end
    end
    return false
end

local function listConfigs()
    ensureFolder(CONFIG_FOLDER)
    local files = listfiles(CONFIG_FOLDER)
    local names = {}
    for _, f in ipairs(files) do
        local n = f:match("([^/\\]+)%.json$")
        if n and n ~= "autoload" then
            table.insert(names, n) [cite: 7]
        end
    end
    table.sort(names)
    return names
end

local function saveAutoload(name)
    ensureFolder(CONFIG_FOLDER)
    if name then
        writefile(AUTOLOAD_PATH, name)
    elseif isfile(AUTOLOAD_PATH) then
        delfile(AUTOLOAD_PATH)
    end
end

local function loadAutoload()
    if isfile(AUTOLOAD_PATH) then
        local name = readfile(AUTOLOAD_PATH)
        if name and name ~= "" then [cite: 8]
            return name
        end
    end
    return nil
end

local function tweenTransparency(frame, target, duration)
    for _, obj in ipairs(frame:GetDescendants()) do
        if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
            TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {TextTransparency = target}):Play() [cite: 16]
        elseif obj:IsA("Frame") or obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
            TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundTransparency = target}):Play()
        end
    end
    TweenService:Create(frame, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundTransparency = target}):Play()
end

local Highlights = {}

local function isOtherPlayerCharacter(char)
    local plr = Players:GetPlayerFromCharacter(char)
    return plr and plr ~= LOCAL_PLAYER
end

local function createHighlightForCharacter(char)
    if not isOtherPlayerCharacter(char) then return end
    if Highlights[char] then return end

    local h = Instance.new("Highlight")
    h.FillTransparency = 1
    h.OutlineTransparency = 0
    h.OutlineColor = Color3.fromRGB(255, 255, 255)
    h.Adornee = char
    h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    h.Parent = ROOT
    Highlights[char] = h
end

local function removeHighlightForCharacter(char) [cite: 58]
    local h = Highlights[char]
    if h then
        h:Destroy()
        Highlights[char] = nil
    end
end

local function refreshHighlights()
    for char, _ in pairs(Highlights) do
        if not char.Parent or not isOtherPlayerCharacter(char) then
            removeHighlightForCharacter(char)
        end
    end
    if not HitboxEnabled then
        for char, _ in pairs(Highlights) do [cite: 59]
            removeHighlightForCharacter(char)
        end
        return
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LOCAL_PLAYER then
            local char = plr.Character
            if char then
                createHighlightForCharacter(char) [cite: 60]
            end
        end
    end
end

Players.PlayerAdded:Connect(function(plr)
    plr.CharacterAdded:Connect(function(char)

        end
    end)
end)

Players.PlayerRemoving:Connect(function(plr)
    if plr ~= LOCAL_PLAYER and plr.Character then

end)
