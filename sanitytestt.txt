                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )

local CONFIG_DIR = "sanity_configs"
local DEFAULT_CONFIG = "sanity_default"

local function safeWrite(path, data)
    if writefile then pcall(writefile, path, data) end
end

local function safeRead(path)
    if isfile and isfile(path) then
        local ok, data = pcall(readfile, path)
        if ok then return data end
    end
    return nil
end

local function listConfigs()
    local names = {}
    if listfiles then
        local ok, files = pcall(listfiles, CONFIG_DIR)
        if ok then
            for _, f in ipairs(files) do
                local n = f:match("([^/\\]+)%.json$")
                if n then table.insert(names, n) end
            end
        end
    end
    table.sort(names)
    return names
end

local CurrentConfig = {
    ShowWatermark = true,
    CustomCursor = false,
    NotificationSide = "Right",
    DPIScale = 0,
    HitboxDebug = false,
    HitboxOthersOnly = true,
}

local function applyConfig(tbl)
    for k,v in pairs(tbl) do
        CurrentConfig[k] = v
    end
    UIScale.Scale = 0.75 + CurrentConfig.DPIScale * 0.75
    Watermark.Visible = CurrentConfig.ShowWatermark
end

local function saveConfig(name)
    if not name or name == "" then name = "config_"..os.time() end
    local path = CONFIG_DIR.."/"..name..".json"
    safeWrite(path, HttpService:JSONEncode(CurrentConfig))
end

local function loadConfig(name)
    local path = CONFIG_DIR.."/"..name..".json"
    local raw = safeRead(path)
    if not raw then return end
    local ok, decoded = pcall(HttpService.JSONDecode, HttpService, raw)
    if ok then applyConfig(decoded) end
end

loadConfig(DEFAULT_CONFIG)

local function makeToggle(parent, text, default)
    local f = Instance.new("Frame", parent)
    f.BackgroundTransparency = 1
    f.Size = UDim2.new(1,-20,0,24)

    local l = Instance.new("TextLabel", f)
    l.BackgroundTransparency = 1
    l.Size = UDim2.new(1,-40,1,0)
    l.Font = Enum.Font.Code
    l.Text = text
    l.TextColor3 = Color3.fromRGB(255,255,255)
    l.TextSize = 14
    l.TextXAlignment = Enum.TextXAlignment.Left

    local b = Instance.new("TextButton", f)
    b.Size = UDim2.fromOffset(32,18)
    b.Position = UDim2.new(1,-32,0.5,0)
    b.AnchorPoint = Vector2.new(1,0.5)
    b.BackgroundColor3 = Color3.fromRGB(255,255,255)
    b.Text = ""
    Instance.new("UICorner", b).CornerRadius = UDim.new(1,0)

    local knob = Instance.new("Frame", b)
    knob.Size = UDim2.fromOffset(14,14)
    knob.Position = UDim2.fromOffset(2,2)
    knob.BackgroundColor3 = Color3.fromRGB(0,0,0)
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1,0)

    local state = default
    local function update()
        if state then
            b.BackgroundColor3 = Color3.fromRGB(255,255,255)
            knob.Position = UDim2.fromOffset(32-14-2,2)
        else
            b.BackgroundColor3 = Color3.fromRGB(100,100,100)
            knob.Position = UDim2.fromOffset(2,2)
        end
    end

    b.MouseButton1Click:Connect(function()
        state = not state
        update()
    end)

    update()

    return {
        Get = function() return state end,
        Set = function(v) state=v update() end,
    }
end
